# ============================================
#  电台:radio_my —— 电台声明与歌曲挂载(_songs.txt)
# ============================================
# 规则:
#  - music_station = "电台名":声明电台,这个名字会原样显示在游戏播放器里
#  - 电台名不能与已有电台重复(同名=合并播放池)
#  - chance.modifier.factor 是播放权重;可以写任意触发条件,
#    条件满足时该 factor 才计入随机抽取池

music_station = "radio_my"

# 无条件曲目:任何时候都可能被抽到
music = {
	song = "示例歌曲一"
	chance = {
		modifier = {
			factor = 1
		}
	}
}

# 带条件曲目:只在战争时计入权重
music = {
	song = "示例歌曲二"
	chance = {
		modifier = {
			factor = 1
			has_war = yes
		}
	}
}
